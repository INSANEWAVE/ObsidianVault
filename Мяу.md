Отлично, программа успешно залита на контроллер STM32, финальный результат завершен, напиши md файл на основе файла  md файла предыдущего курсового проекта, суть и основные расчёты для проекта точно такие же как в оригинальном документе, тебе необходимо переписать его сохранив необходимые теоретические формулы, но перепиши блок чтобы вместо AHDL речь шла про STM32

Вот рабочий код системы который в итоге был применён в проекте, Почисти код и уменьши в нём след нейросетей, он должен выглядеть чуточку более рукописным. Обязательно внеси в курсовой проект листинг этого кода:

```
/*
 * STM32F407VET6 + FNIRSI DSO-510
 * Аппаратный тест генерации сигналов
 *
 * ================================================================
 * НАЗНАЧЕНИЕ
 * ================================================================
 *
 * Эта программа предназначена НЕ для демонстрации сложного
 * приложения, а для проверки именно аппаратных возможностей
 * STM32F407VET6 по генерации сигналов, которые удобно измерять
 * осциллографом.
 *
 * Выводы:
 *
 *   PA4 -> DAC1_OUT1 -> синус 500 Гц
 *   PA6 -> TIM3_CH1  -> PWM 1 кГц, 50%
 *   PA7 -> TIM14_CH1 -> PWM 100 Гц, 50%
 *
 * Основная идея:
 *
 *   TIM6 -> TRGO -> DAC trigger
 *             |
 *             +-> DMA -> следующий отсчёт синуса
 *
 * Поэтому генерация синуса не зависит от времени выполнения
 * обработчика прерывания Cortex-M4.
 *
 * Цифровые сигналы PA6/PA7 также формируются аппаратными
 * таймерами, без gpio_toggle() и без программных задержек.
 *
 * ================================================================
 * ТРЕБОВАНИЯ К ПЛАТЕ
 * ================================================================
 *
 * Код рассчитан на:
 *
 *   STM32F407VET6
 *   внешний HSE = 8 МГц
 *   SYSCLK = 168 МГц
 *
 * ВАЖНО:
 * Если на твоей конкретной плате стоит кварц НЕ 8 МГц,
 * clock_setup() необходимо изменить.
 *
 * ================================================================
 * ЧАСТОТЫ
 * ================================================================
 *
 * DAC:
 *   Fs = 10 кГц
 *   20 отсчётов на период
 *   Fsignal = 10000 / 20 = 500 Гц
 *
 * PA6:
 *   TIM3_CH1 = 1 кГц
 *   duty = 50%
 *
 * PA7:
 *   TIM14_CH1 = 100 Гц
 *   duty = 50%
 *
 * ================================================================
 * ПОДКЛЮЧЕНИЕ К FNIRSI
 * ================================================================
 *
 * Земля платы STM32 и земля осциллографа должны быть соединены.
 *
 * Канал осциллографа 1 -> PA4
 * Канал осциллографа 2 -> PA6
 * Канал осциллографа 3 -> PA7
 *
 * Если у твоего DSO-510 физически доступен только один канал,
 * сигналы можно проверять по очереди.
 *
 * Для DAC используй DC coupling.
 *
 * ================================================================
 */

#include <stdint.h>

#include <libopencm3/stm32/rcc.h>
#include <libopencm3/stm32/gpio.h>
#include <libopencm3/stm32/timer.h>
#include <libopencm3/stm32/dac.h>
#include <libopencm3/stm32/dma.h>


/* -------------------------------------------------------------------------- */
/* Настройки теста                                                            */
/* -------------------------------------------------------------------------- */

/*
 * Частота обновления DAC.
 *
 * 10 кГц / 20 отсчётов = 500 Гц.
 */
#define DAC_SAMPLE_RATE_HZ     10000U

/*
 * Размер таблицы синуса.
 */
#define SINE_TABLE_SIZE        20U

/*
 * Проверяемые частоты цифровых сигналов.
 */
#define PA6_FREQUENCY_HZ       1000U
#define PA7_FREQUENCY_HZ       100U


/* -------------------------------------------------------------------------- */
/* Таблица синуса                                                             */
/* -------------------------------------------------------------------------- */

/*
 * Синус:
 *
 *   центр = 2048
 *   амплитуда = 1500 отсчётов
 *
 * При VREF = 3.3 В:
 *
 *   центр ≈ 1.65 В
 *   амплитуда ≈ 1.21 В
 *   размах ≈ 2.42 Vpp
 *
 * Значения специально оставлены 12-битными и непосредственно
 * подходят для DAC1 channel 1.
 *
 * 20 отсчётов на период дают:
 *
 *   10000 / 20 = 500 Гц
 *
 * Ступенчатость при увеличении масштаба осциллографа здесь
 * является нормальным следствием всего 20 отсчётов на период.
 */
static const uint16_t sine_500hz[SINE_TABLE_SIZE] = {
    2048,
    2512,
    2930,
    3262,
    3475,
    3548,
    3475,
    3262,
    2930,
    2512,
    2048,
    1584,
    1166,
    834,
    621,
    548,
    621,
    834,
    1166,
    1584
};


/* -------------------------------------------------------------------------- */
/* Тактирование                                                               */
/* -------------------------------------------------------------------------- */

static void clock_setup(void)
{
    /*
     * Предполагаем внешний кварц/генератор 8 МГц.
     *
     * Получаем:
     *
     *   SYSCLK = 168 МГц
     *   APB1   = 42 МГц
     *
     * Для таймеров APB1 при делителе APB1 != 1 действует
     * умножение таймерной частоты на 2:
     *
     *   TIMxCLK = 84 МГц
     */
    rcc_clock_setup_pll(
        &rcc_hse_8mhz_3v3[RCC_CLOCK_3V3_168MHZ]
    );

    /*
     * Включаем тактирование всех используемых периферий.
     */
    rcc_periph_clock_enable(RCC_GPIOA);

    rcc_periph_clock_enable(RCC_DAC);

    rcc_periph_clock_enable(RCC_DMA1);

    rcc_periph_clock_enable(RCC_TIM3);
    rcc_periph_clock_enable(RCC_TIM6);
    rcc_periph_clock_enable(RCC_TIM14);
}


/* -------------------------------------------------------------------------- */
/* GPIO                                                                        */
/* -------------------------------------------------------------------------- */

static void gpio_setup(void)
{
    /*
     * ----------------------------------------------------------------------
     * PA4 -> DAC1_OUT1
     * ----------------------------------------------------------------------
     *
     * Для DAC вывод должен находиться в аналоговом режиме.
     */
    gpio_mode_setup(
        GPIOA,
        GPIO_MODE_ANALOG,
        GPIO_PUPD_NONE,
        GPIO4
    );

    /*
     * ----------------------------------------------------------------------
     * PA6 -> TIM3_CH1
     * PA7 -> TIM14_CH1
     * ----------------------------------------------------------------------
     *
     * Оба вывода переводим в Alternate Function.
     */
    gpio_mode_setup(
        GPIOA,
        GPIO_MODE_AF,
        GPIO_PUPD_NONE,
        GPIO6 | GPIO7
    );

    /*
     * PA6:
     *
     *   TIM3_CH1 = AF2
     */
    gpio_set_af(
        GPIOA,
        GPIO_AF2,
        GPIO6
    );

    /*
     * PA7:
     *
     *   TIM14_CH1 = AF9
     */
    gpio_set_af(
        GPIOA,
        GPIO_AF9,
        GPIO7
    );

    /*
     * Для цифровых сигналов используем push-pull и высокую
     * скорость переключения.
     *
     * Это помогает получить более резкие фронты на осциллографе.
     */
    gpio_set_output_options(
        GPIOA,
        GPIO_OTYPE_PP,
        GPIO_OSPEED_100MHZ,
        GPIO6 | GPIO7
    );
}


/* -------------------------------------------------------------------------- */
/* DAC                                                                         */
/* -------------------------------------------------------------------------- */

static void dac_setup(void)
{
    /*
     * На всякий случай начинаем с выключенного канала.
     */
    dac_disable(DAC1, DAC_CHANNEL1);

    /*
     * Отключаем встроенную генерацию waveform.
     *
     * Нам нужна собственная таблица синуса через DMA.
     */
    dac_disable_waveform_generation(
        DAC1,
        DAC_CHANNEL1
    );

    /*
     * Включаем выходной буфер DAC.
     *
     * Для непосредственного подключения осциллографического
     * входа с большим сопротивлением это наиболее удобный режим.
     */
    dac_buffer_enable(
        DAC1,
        DAC_CHANNEL1
    );

    /*
     * ----------------------------------------------------------------------
     * Выбор TIM6 TRGO как источника триггера DAC.
     * ----------------------------------------------------------------------
     *
     * В STM32F407 поле TSEL1 после сброса содержит значение SW trigger.
     *
     * Функция dac_set_trigger_source() в libopencm3 только добавляет
     * соответствующие биты через OR, поэтому для перехода на TIM6
     * надёжнее явно очистить поле TSEL1.
     */
    DAC_CR(DAC1) &= ~(0x7U << DAC_CR_TSEL1_SHIFT);

    /*
     * Для STM32F4:
     *
     *   TSEL1 = 000 -> TIM6 TRGO
     */
    DAC_CR(DAC1) |= DAC_CR_TSEL1_T6;

    /*
     * Разрешаем внешний триггер DAC.
     */
    dac_trigger_enable(
        DAC1,
        DAC_CHANNEL1
    );

    /*
     * Очень важный момент:
     *
     * Перед запуском таймера загружаем в DHR последний отсчёт.
     *
     * На первом TIM6 TRGO:
     *
     *   DHR -> DOR = последний отсчёт
     *   DMA -> DHR = первый отсчёт
     *
     * Затем получается непрерывная последовательность:
     *
     *   last -> 0 -> 1 -> 2 -> ... -> last -> 0 -> ...
     *
     * То есть первый цикл не содержит лишнего нулевого значения.
     */
    dac_load_data_buffer_single(
        DAC1,
        sine_500hz[SINE_TABLE_SIZE - 1U],
        DAC_ALIGN_RIGHT12,
        DAC_CHANNEL1
    );

    /*
     * Разрешаем DMA-запросы DAC.
     *
     * В STM32F407 DAC1 использует:
     *
     *   DMA1 Stream 5
     *   Channel 7
     */
    dac_dma_enable(
        DAC1,
        DAC_CHANNEL1
    );

    /*
     * Наконец включаем сам DAC.
     */
    dac_enable(
        DAC1,
        DAC_CHANNEL1
    );
}


/* -------------------------------------------------------------------------- */
/* DMA для DAC                                                                */
/* -------------------------------------------------------------------------- */

static void dac_dma_setup(void)
{
    /*
     * DAC1:
     *
     *   DMA1
     *   Stream 5
     *   Channel 7
     *
     * Перед настройкой поток сбрасываем.
     */
    dma_stream_reset(
        DMA1,
        DMA_STREAM5
    );

    /*
     * Выбираем канал 7.
     */
    dma_channel_select(
        DMA1,
        DMA_STREAM5,
        DMA_SxCR_CHSEL_7
    );

    /*
     * Направление:
     *
     *   память -> периферия
     */
    dma_set_transfer_mode(
        DMA1,
        DMA_STREAM5,
        DMA_SxCR_DIR_MEM_TO_PERIPHERAL
    );

    /*
     * DAC работает с 12-битными значениями.
     *
     * Используем 16-битный DMA transfer.
     */
    dma_set_peripheral_size(
        DMA1,
        DMA_STREAM5,
        DMA_SxCR_PSIZE_16BIT
    );

    dma_set_memory_size(
        DMA1,
        DMA_STREAM5,
        DMA_SxCR_MSIZE_16BIT
    );

    /*
     * Высокий приоритет для DMA.
     *
     * Для данного теста это не строго необходимо, но позволяет
     * уменьшить вероятность задержки DMA-запроса при добавлении
     * других периферий в будущем.
     */
    dma_set_priority(
        DMA1,
        DMA_STREAM5,
        DMA_SxCR_PL_VERY_HIGH
    );

    /*
     * Адрес DAC DHR12R1.
     *
     * DMA будет писать очередное 16-битное значение непосредственно
     * в регистр хранения данных DAC1 channel 1.
     */
    dma_set_peripheral_address(
        DMA1,
        DMA_STREAM5,
        (uint32_t)&DAC_DHR12R1(DAC1)
    );

    /*
     * Начальный адрес таблицы синуса.
     */
    dma_set_memory_address(
        DMA1,
        DMA_STREAM5,
        (uint32_t)sine_500hz
    );

    /*
     * Всего передаём 20 значений.
     */
    dma_set_number_of_data(
        DMA1,
        DMA_STREAM5,
        SINE_TABLE_SIZE
    );

    /*
     * После каждого DMA-переноса увеличиваем адрес в памяти.
     *
     * Поэтому идём:
     *
     *   sine[0]
     *   sine[1]
     *   ...
     *   sine[19]
     */
    dma_enable_memory_increment_mode(
        DMA1,
        DMA_STREAM5
    );

    /*
     * Адрес периферии всегда один и тот же:
     *
     * DAC_DHR12R1
     */
    dma_disable_peripheral_increment_mode(
        DMA1,
        DMA_STREAM5
    );

    /*
     * После последнего элемента автоматически начинаем таблицу
     * заново.
     */
    dma_enable_circular_mode(
        DMA1,
        DMA_STREAM5
    );

    /*
     * Direct mode:
     *
     * DMA сразу выполняет передачу по запросу DAC,
     * без необходимости использовать FIFO.
     */
    dma_enable_direct_mode(
        DMA1,
        DMA_STREAM5
    );

    /*
     * Теперь DMA-канал готов.
     */
    dma_enable_stream(
        DMA1,
        DMA_STREAM5
    );
}


/* -------------------------------------------------------------------------- */
/* TIM6 -> DAC sample clock                                                   */
/* -------------------------------------------------------------------------- */

static void tim6_setup(void)
{
    const uint32_t timer_clock_hz =
        rcc_get_timer_clk_freq(TIM6);

    /*
     * Используем удобную внутреннюю частоту счётчика 1 МГц.
     *
     * Например:
     *
     *   84 МГц / 84 = 1 МГц
     */
    const uint32_t counter_clock_hz = 1000000U;

    /*
     * Делитель до 1 МГц.
     */
    const uint32_t prescaler =
        (timer_clock_hz / counter_clock_hz) - 1U;

    /*
     * При 1 МГц:
     *
     *   1000000 / 10000 = 100 тиков
     *
     * Поэтому ARR = 99.
     */
    const uint32_t period =
        (counter_clock_hz / DAC_SAMPLE_RATE_HZ) - 1U;

    /*
     * Базовая конфигурация TIM6.
     */
    timer_set_mode(
        TIM6,
        TIM_CR1_CKD_CK_INT,
        TIM_CR1_CMS_EDGE,
        TIM_CR1_DIR_UP
    );

    timer_continuous_mode(TIM6);

    timer_set_prescaler(
        TIM6,
        prescaler
    );

    timer_set_period(
        TIM6,
        period
    );

    /*
     * Формируем TIM6 TRGO на каждом Update Event.
     *
     * Именно этим сигналом будет запускаться DAC.
     */
    timer_set_master_mode(
        TIM6,
        TIM_CR2_MMS_UPDATE
    );

    /*
     * Update event нужен для загрузки PSC/ARR.
     *
     * ВАЖНО:
     * Мы пока ещё не запускаем счётчик.
     */
    timer_generate_event(
        TIM6,
        TIM_EGR_UG
    );

    /*
     * После UG заново выставляем MMS.
     *
     * Это гарантирует, что служебный UG при конфигурации
     * не использовался как первый DAC trigger.
     */
    timer_set_master_mode(
        TIM6,
        TIM_CR2_MMS_UPDATE
    );
}


/* -------------------------------------------------------------------------- */
/* Аппаратный PWM                                                             */
/* -------------------------------------------------------------------------- */

static void pwm_setup(
    uint32_t timer,
    uint32_t frequency_hz
)
{
    /*
     * Получаем реальную частоту конкретного таймера.
     *
     * Это лучше, чем зашивать 84 МГц вручную:
     * если конфигурация тактирования изменится, расчёт всё ещё
     * останется корректным.
     */
    const uint32_t timer_clock_hz =
        rcc_get_timer_clk_freq(timer);

    /*
     * Делаем частоту счётчика 1 МГц.
     *
     * Тогда:
     *
     *   1 тик = 1 мкс
     */
    const uint32_t counter_clock_hz = 1000000U;

    const uint32_t prescaler =
        (timer_clock_hz / counter_clock_hz) - 1U;

    /*
     * ARR рассчитывается из требуемой частоты PWM.
     *
     * Для 1 кГц:
     *
     *   1 000 000 / 1000 = 1000
     *   ARR = 999
     *
     * Для 100 Гц:
     *
     *   1 000 000 / 100 = 10000
     *   ARR = 9999
     */
    const uint32_t period =
        (counter_clock_hz / frequency_hz) - 1U;

    /*
     * 50 % заполнение.
     *
     * Например для 1 кГц:
     *
     *   ARR = 999
     *   CCR = 500
     */
    const uint32_t compare =
        (period + 1U) / 2U;

    timer_set_mode(
        timer,
        TIM_CR1_CKD_CK_INT,
        TIM_CR1_CMS_EDGE,
        TIM_CR1_DIR_UP
    );

    timer_continuous_mode(timer);

    timer_set_prescaler(
        timer,
        prescaler
    );

    timer_set_period(
        timer,
        period
    );

    /*
     * PWM mode 1:
     *
     * сигнал высокий при CNT < CCR1,
     * затем низкий до конца периода.
     */
    timer_set_oc_mode(
        timer,
        TIM_OC1,
        TIM_OCM_PWM1
    );

    /*
     * Активный уровень — высокий.
     */
    timer_set_oc_polarity_high(
        timer,
        TIM_OC1
    );

    /*
     * Включаем выход канала.
     */
    timer_enable_oc_output(
        timer,
        TIM_OC1
    );

    /*
     * Для TIM3 функция timer_set_oc_value() доступна,
     * но в libopencm3 она имеет ограничения для некоторых
     * таймеров, включая TIM14.
     *
     * Поэтому здесь напрямую используем регистр CCR1.
     *
     * Для STM32F407 это штатный регистр TIMx_CCR1.
     */
    TIM_CCR1(timer) = compare;

    /*
     * Применяем настройки.
     */
    timer_generate_event(
        timer,
        TIM_EGR_UG
    );

    /*
     * Запускаем таймер.
     */
    timer_enable_counter(timer);
}


/* -------------------------------------------------------------------------- */
/* TIM3: PA6 -> 1 кГц                                                         */
/* -------------------------------------------------------------------------- */

static void tim3_setup(void)
{
    /*
     * PA6 уже переведён в:
     *
     *   Alternate Function 2
     *   TIM3_CH1
     *
     * Здесь создаём чистый аппаратный PWM 1 кГц / 50%.
     */
    pwm_setup(
        TIM3,
        PA6_FREQUENCY_HZ
    );
}


/* -------------------------------------------------------------------------- */
/* TIM14: PA7 -> 100 Гц                                                        */
/* -------------------------------------------------------------------------- */

static void tim14_setup(void)
{
    /*
     * PA7:
     *
     *   Alternate Function 9
     *   TIM14_CH1
     *
     * TIM14 формирует 100 Гц независимо от TIM3.
     */
    pwm_setup(
        TIM14,
        PA7_FREQUENCY_HZ
    );
}


/* -------------------------------------------------------------------------- */
/* Запуск TIM6                                                                 */
/* -------------------------------------------------------------------------- */

static void tim6_start(void)
{
    /*
     * На этом этапе уже должны быть настроены:
     *
     *   DAC
     *   DMA
     *   TIM6
     *
     * Первый период начнётся с текущего состояния CNT = 0.
     */
    timer_enable_counter(TIM6);
}


/* -------------------------------------------------------------------------- */
/* main                                                                        */
/* -------------------------------------------------------------------------- */

int main(void)
{
    /*
     * 1. Настраиваем тактирование.
     */
    clock_setup();

    /*
     * 2. Настраиваем выводы.
     */
    gpio_setup();

    /*
     * 3. Настраиваем DAC.
     */
    dac_setup();

    /*
     * 4. Настраиваем DMA, который будет циклически передавать
     *    таблицу синуса в DAC.
     */
    dac_dma_setup();

    /*
     * 5. Настраиваем TIM6.
     *
     * TIM6 определяет точный момент каждого DAC sample.
     */
    tim6_setup();

    /*
     * 6. Аппаратный PWM 1 кГц на PA6.
     */
    tim3_setup();

    /*
     * 7. Аппаратный PWM 100 Гц на PA7.
     */
    tim14_setup();

    /*
     * 8. Только теперь запускаем sample timer DAC.
     */
    tim6_start();

    /*
     * После этого вся генерация выполняется периферией:
     *
     *   TIM6 -> DAC TRGO -> DAC
     *                 |
     *                 +-> DMA -> таблица
     *
     *   TIM3  -> PA6
     *
     *   TIM14 -> PA7
     *
     * CPU больше не обязан обслуживать генерацию сигналов.
     */
    while (1) {
        /*
         * Программа ничего не делает.
         *
         * Никаких задержек и программных toggle здесь нет.
         */
        __asm__ volatile ("nop");
    }

    return 0;
}
```
